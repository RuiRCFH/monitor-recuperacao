Função para detetar o horário e focar no turno certo ao abrir function focarTurnoAtual() { const hora = new Date().getHours(); let idTurno = ""; if (hora >= 5 && hora < 12) idTurno = "collapseManha"; else if (hora >= 12 && hora < 18) idTurno = "collapseTarde"; else idTurno = "collapseNoite"; // Abre o turno atual automaticamente const elemento = document.getElementById(idTurno); if (elemento) { new bootstrap.Collapse(elemento, { show: true }); } } // 2. Função para saltar para o próximo turno após clicar function proximoTurno(atual) { const ordem = ["Manha", "Tarde", "Noite"]; const index = ordem.indexOf(atual); if (index >= 0 && index < ordem.length - 1) { const proximoID = "collapse" + ordem[index + 1]; const atualID = "collapse" + ordem[index]; // Pequena pausa para o paciente ver que marcou setTimeout(() => { // Fecha o atual e abre o próximo new bootstrap.Collapse(document.getElementById(atualID)).hide(); new bootstrap.Collapse(document.getElementById(proximoID)).show(); // Faz um scroll suave para o novo turno document.getElementById(proximoID).scrollIntoView({ behavior: 'smooth' }); }, 500); } } // 3. Chamar a função ao carregar a página window.onload = function() { focarTurnoAtual(); };
